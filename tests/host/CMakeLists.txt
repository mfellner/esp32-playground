cmake_minimum_required(VERSION 3.16)
project(platform_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SANITIZE "Address/undefined sanitizers" ON)
if(SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-Wall -Wextra)
set(ROOT ${CMAKE_CURRENT_LIST_DIR}/../..)
enable_testing()
add_executable(orientation_tests orientation_tests.cpp ${ROOT}/components/board/orientation.cpp)
target_include_directories(orientation_tests PRIVATE ${ROOT}/components/board/include)
add_test(NAME orientation_tests COMMAND orientation_tests)
add_executable(slot_model_tests slot_model_tests.cpp ${ROOT}/firmware/launcher/main/slot_model.cpp)
target_include_directories(slot_model_tests PRIVATE ${ROOT}/firmware/launcher/main)
add_test(NAME slot_model_tests COMMAND slot_model_tests)
